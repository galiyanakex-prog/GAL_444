    1 Host vps
    2     HostName 91.188.212.77
    3    User t
    4     IdentityFile ~/.ssh/id_ed25519
    5     ServerAliveInterval 20
    6    ServerAliveCountMax 3
    7
    8 Host m45
    9     HostName 127.0.0.1
   10     Port 222
   11     User u
   12     ProxyJump vps
   13     ServerAliveInterval 20
   14     ServerAliveCountMax 3

   1 # на VPS по паролю (пароль юзера t)

   3
   4 # на хост по паролю через туннель (пароль юзера u)
   5 ssh -o PubkeyAuthentication=no -o PreferredAuthentications=password -J vps -p 222 u@127.0.1
